<script>
  function animateValue(id, start, end, duration) {
    const el = document.getElementById(id);
    let startTimestamp = null;
    const step = (timestamp) => {
      if (!startTimestamp) startTimestamp = timestamp;
      const progress = Math.min((timestamp - startTimestamp) / duration, 1);
      el.innerHTML = Math.floor(progress * (end - start) + start);
      if (progress < 1) {
        window.requestAnimationFrame(step);
      }
    };
    window.requestAnimationFrame(step);
  }
  // Run loop hook instances
  animateValue('counter-target-node', 0, 150, 2000);
</script>

<script>
  function accordionToggle(btn) {
    const pane = btn.nextElementSibling;
    const icon = btn.querySelector('.icon-indicator');
    const isClosed = pane.classList.contains('hidden');
    
    // Close remaining sister instances
    document.querySelectorAll('.content-pane').forEach(p => p.classList.add('hidden'));
    document.querySelectorAll('.icon-indicator').forEach(i => i.textContent = "+");
    
    if(isClosed) {
      pane.classList.remove('hidden');
      icon.textContent = "−";
    }
  }
</script>
