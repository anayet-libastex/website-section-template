10. Resume Upload (Drag & Drop Active JS)

Interactive drag zone area handling telemetry upload configurations dynamically.

📄

Drag and drop profile asset document

Accepts PDF, DOCX formats down up to 10MB limits

<!-- 10. RESUME UPLOAD (DRAG & DROP) -->
<div id="dragDropUploadZone" class="max-w-md mx-auto border-2 border-dashed border-zinc-800 bg-zinc-900/50 p-8 rounded-2xl text-center space-y-2 cursor-pointer hover:border-violet-500/50 hover:bg-zinc-900 transition-all">
  <span class="text-2xl block">📄</span>
  <h4 class="font-display font-bold text-sm text-white" id="uploadZoneTitleText">Drag and drop profile asset document</h4>
  <p class="text-zinc-500 text-[11px] font-mono">Accepts PDF, DOCX formats down up to 10MB limits</p>
  <input type="file" id="hiddenFileInputElement" class="hidden" accept=".pdf,.docx">
</div>

<script>
  const zone = document.getElementById('dragDropUploadZone');
  const text = document.getElementById('uploadZoneTitleText');
  const inp = document.getElementById('hiddenFileInputElement');
  
  zone.addEventListener('click', () => inp.click());
  zone.addEventListener('dragover', (e) => { e.preventDefault(); zone.classList.add('border-violet-500'); });
  zone.addEventListener('dragleave', () => { zone.classList.remove('border-violet-500'); });
  zone.addEventListener('drop', (e) => {
    e.preventDefault();
    zone.classList.remove('border-violet-500');
    if(e.dataTransfer.files.length) {
      text.innerText = "Captured: " + e.dataTransfer.files[0].name;
      text.classList.add('text-violet-400');
    }
  });
  inp.addEventListener('change', () => {
    if(inp.files.length) {
      text.innerText = "Attached: " + inp.files[0].name;
      text.classList.add('text-violet-400');
    }
  });
</script>
